cmake_minimum_required(VERSION 3.28)
project(BananaPadShellTools LANGUAGES C CXX)
set(BANANAPAD_SOURCE "" CACHE PATH "Verified pinned DK64 source")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(rt64 "${BANANAPAD_SOURCE}/lib/rt64")
set(spirv "${rt64}/src/contrib/spirv-cross")
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_UTIL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory("${spirv}" spirv-cross EXCLUDE_FROM_ALL)
add_executable(file_to_c "${rt64}/src/tools/file_to_c/file_to_c.cpp")
target_compile_features(file_to_c PRIVATE cxx_std_17)
add_executable(spirv_cross_msl "${rt64}/src/tools/spirv_cross_msl/spirv_cross_msl.cpp")
target_compile_features(spirv_cross_msl PRIVATE cxx_std_17)
target_include_directories(spirv_cross_msl PRIVATE "${spirv}")
target_link_libraries(spirv_cross_msl PRIVATE spirv-cross-core spirv-cross-msl)
